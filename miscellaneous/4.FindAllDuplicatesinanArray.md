# Step 8 — Swift Code

```swift
func findDuplicates(_ nums: [Int]) -> [Int] {

    var visited = Set<Int>()
    var reported = Set<Int>()
    var result: [Int] = []

    for num in nums {

        if visited.contains(num) {

            if !reported.contains(num) {
                result.append(num)
                reported.insert(num)
            }

        } else {
            visited.insert(num)
        }
    }

    return result
}
```
